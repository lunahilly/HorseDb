db.sql :
{
create type horse_weight_state as enum (
    'none',
    'good_weight',
    'underweight',
    'overweight'
    );

create type state as enum (
    'none',
    'active',
    'inactive',
    'pension',
    'under_care'
    );

create type horse_sex as enum (
    'none',
    'mare',
    'stallion',
    'gelding'
    );

create type human_sex as enum (
    'none',
    'female',
    'male'
    );

create type appointment_type as enum (
    'none',
    'veterinary',
    'physio',
    'farrier',
    'other'
    );

CREATE TABLE owners
(
    id          serial    not null primary key,
    given_name  text      not null,
    family_name text      not null,
    birth_date  date,
    sex         human_sex not null,
    email       text      not null,
    phone       varchar(13)
);

CREATE TABLE horse_races
(
    id   serial not null primary key,
    name text   not null
);

CREATE TABLE horses
(
    id           serial      not null primary key,
    owner_id     int         not null references owners,
    name         varchar(32) not null,
    full_name    text        not null,
    chip         varchar(15) not null unique,
    sex          horse_sex   not null,
    birth_date   date,
    race_id      int         not null references horse_races,
    color        varchar(32),
    height       int,
    weight       int,
    weight_state horse_weight_state,
    state        state       not null
);

CREATE TABLE horse_images
(
    id       serial not null primary key,
    horse_id int    not null references horses,
    image    text   not null
);

CREATE TABLE horse_comments
(
    id       serial not null primary key,
    horse_id int    not null references horses,
    date     date   not null default current_date,
    comment  text   not null
);

CREATE TABLE appointments
(
    id          serial           not null primary key,
    horse_id    int              not null references horses,
    type        appointment_type not null,
    date        timestamptz      not null,
    description text
)
}
